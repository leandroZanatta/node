# O "pom" de cada pacote: o que ele PODE importar. Nada mais.
#
# Isto é o contrato que um módulo Maven imporia, num projeto que termina em um
# processo só e não ganharia nada com multi-módulo. Acrescentar uma aresta aqui
# é um ato consciente — do mesmo tamanho de editar um pom — e `ArquiteturaTest`
# reprova o build se o código importar o que não está declarado.
#
# A ordem abaixo é a ordem das dependências: quem está em cima não conhece quem
# está embaixo. O grafo tem de ser acíclico.
#
# CICATRIZ: `TabelaDeRotas` nasceu em `descoberta` e criou o ciclo
# descoberta <-> sessao — a sessão precisava dela para responder FindNode, e a
# descoberta precisava da sessão. Passou despercebido porque pacote não impõe
# nada. A tabela é estrutura do protocolo, não política de varredura: desceu
# para `kademlia`.

exception:
vo:
rlp:        exception
util:       exception
cadeia:     vo
kademlia:   vo, util
rede:       vo, rlp, util, exception
sessao:     vo, util, cadeia, kademlia, rede, exception
descoberta: vo, util, kademlia, rede, sessao, exception
api:        ensaio
repository: vo, util, exception, catalogo
nucleo:
catalogo:   vo, util, nucleo
sincronizacao: vo, util, nucleo, catalogo, rede, sessao, exception, repository
sondagem:   util, nucleo, catalogo
mempool:    vo, util, rlp, sessao, nucleo, catalogo
ensaio:     util, vo, nucleo, catalogo, rlp, mempool, sincronizacao

# A raiz compõe o node inteiro, então é a última: depende de todos e ninguém depende dela.
raiz:       vo, util, cadeia, kademlia, rede, sessao, descoberta, repository, catalogo, sincronizacao, sondagem, mempool, ensaio
